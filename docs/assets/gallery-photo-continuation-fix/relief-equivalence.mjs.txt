// Identical-output check: the shipped reliefTexture() (5fb9801) and the banded one in the working tree, run on the same
// ImageBitmaps in one Chromium page, must produce byte-identical relief data and the same texture settings.
// Inputs: every photo fixture decoded the way acquirePhoto decodes (flipY, square resize at 1024 / 512 / 256), plus seeded
// synthetic images (noise with varied alpha, gradients, flat, a transparent disc, non-square and sub-512 sizes).
// usage, from the worktree root: node outputs/gpc/relief-equivalence.mjs
import fs from 'node:fs';
import {createRequire} from 'node:module';
import {execFileSync} from 'node:child_process';
const {chromium}=createRequire(import.meta.url)(process.env.PLAYWRIGHT_MODULE||'playwright');
const BASE='5fb9801',origin='https://equiv.test';
const exportRelief=src=>src+'\nexport {reliefTexture};\n';
const files={'/old/photo.mjs':exportRelief(execFileSync('git',['show',BASE+':public/disc3d/photo.mjs'],{encoding:'utf8'})),'/new/photo.mjs':exportRelief(fs.readFileSync(process.env.RELIEF_NEW||'public/disc3d/photo.mjs','utf8'))};
const fixtures=['stamped-disc.webp','disc-photo.jpg','disc-photo.png','crooked-stamp.jpg','photo-edge-clean.jpg','photo-edge-wraith.jpg'];
const browser=await chromium.launch({headless:true,args:['--enable-gpu']});
try{
 const page=await browser.newPage();
 await page.route(origin+'/**',route=>{
  const p=new URL(route.request().url()).pathname;
  if(p==='/')return route.fulfill({contentType:'text/html',body:'<!doctype html><script type="importmap">{"imports":{"three":"/bag3d/vendor/three/three.module.min.js"}}</script>'});
  if(files[p])return route.fulfill({contentType:'application/javascript',body:files[p]});
  if(p.startsWith('/fixtures/'))return route.fulfill({body:fs.readFileSync('tests/fixtures/'+p.slice(10))});
  const local=/^\/(old|new)\//.test(p)?'public/disc3d/'+p.slice(5):'public'+p;
  return route.fulfill({contentType:'application/javascript',body:fs.readFileSync(local)});
 });
 await page.goto(origin+'/');
 const result=await page.evaluate(async fixtures=>{
  const [{reliefTexture:before},{reliefTexture:after}]=await Promise.all([import('/old/photo.mjs'),import('/new/photo.mjs')]);
  const inputs=[];
  for(const f of fixtures){const blob=await (await fetch('/fixtures/'+f)).blob();for(const size of [1024,512,256])inputs.push([`${f}@${size}`,await createImageBitmap(blob,{imageOrientation:'flipY',resizeWidth:size,resizeHeight:size,resizeQuality:'high'})]);}
  let seed=7;const rand=()=>(seed=(seed*1103515245+12345)>>>0)/2**32;
  const synth=async(name,w,h,paint)=>{const c=new OffscreenCanvas(w,h),g=c.getContext('2d'),img=g.createImageData(w,h);for(let y=0;y<h;y++)for(let x=0;x<w;x++){const o=4*(y*w+x),[r,gg,b,a]=paint(x/w,y/h);img.data.set([r,gg,b,a],o);}g.putImageData(img,0,0);inputs.push([name,await createImageBitmap(c)]);};
  await synth('noise-varied-alpha 1024',1024,1024,()=>[rand()*256,rand()*256,rand()*256,rand()<.7?255:rand()*256]);
  await synth('gradient 512',512,512,(x,y)=>[x*255,y*255,(x+y)*127,255]);
  await synth('flat 1024',1024,1024,()=>[200,40,90,255]);
  await synth('transparent disc 600',600,600,(x,y)=>{const r=Math.hypot(x-.5,y-.5);return [rand()*80+100,120,30,r<.4?255:0];});
  await synth('non-square 700x300',700,300,(x,y)=>[(x*13%1)*255,(y*7%1)*255,rand()*255,255]);
  await synth('small 37',37,37,()=>[rand()*256,rand()*256,rand()*256,255]);
  await synth('stamp-like 1024',1024,1024,(x,y)=>{const ink=Math.abs(Math.sin(x*40)*Math.cos(y*25))>.8;return ink?[20,20,30,255]:[230,220,200,255];});
  const rows=[];
  for(const [name,image] of inputs){
   const a=before(image),b=await after(image,()=>true),da=a.image.data,db=b.image.data;
   let diff=0;for(let i=0;i<Math.max(da.length,db.length);i++)if(da[i]!==db[i])diff++;
   const same=k=>a[k]===b[k];
   rows.push({name,n:a.image.width,bytes:da.length,diff,settings:['format','type','minFilter','magFilter','generateMipmaps'].every(same)&&a.image.width===b.image.width&&a.image.height===b.image.height});
  }
  return rows;
 },fixtures);
 for(const r of result)console.log(`${r.diff===0&&r.settings?'identical':'DIFFERENT'}  ${r.name}  n=${r.n} bytes=${r.bytes} differing=${r.diff} settings=${r.settings}`);
 const bad=result.filter(r=>r.diff||!r.settings);console.log(bad.length?`FAIL ${bad.length} of ${result.length}`:`PASS ${result.length} of ${result.length} inputs byte-identical`);
 process.exitCode=bad.length?1:0;
}finally{await browser.close();}
